/**
 * Quote Desk – Gmail Connector
 *
 * ONE-CLICK MODE (recommended): the admin deploys this project once as a Web app
 *   (Execute as: User accessing the web app · Who has access: Anyone with a Google account)
 *   and saves its URL in the API's Script properties as CONNECTOR_APP_URL.
 *   Each user then clicks "Connect Gmail" in Quote Desk > Settings – no copying, no code.
 *
 * MANUAL MODE (fallback): make a copy, paste your key into CONNECT_KEY, run setup().
 *
 * It only reads deal-like emails. It never sends or deletes email.
 */

var CONNECT_KEY = 'qd_9281d79bc4424ccfb66e00deba54e315';   // manual mode only – leave as is for one-click
var API_URL = 'https://script.google.com/macros/s/AKfycbxP_tsg3zjAldnckfjTpGHccfqUMGtpI-scke8g4DZ4C6_g9gweZb7zf4VphhkKsl7e/exec';     // the Quote Desk API /exec URL

// Which emails to send to Quote Desk. Only deal-like emails – not your whole mailbox.
var SEARCH = 'newer_than:3d -in:chats -in:spam -in:trash ' +
  '(has:attachment OR subject:(quote OR quotation OR price OR pricing OR proposal OR BOQ OR SoW OR RFQ OR "||"))';
var SKIP_SUBJECT = /\b(invoice|receipt|statement|payment|delivery note|payslip|newsletter|unsubscribe|order confirmation)\b/i;
var ATTACH_OK = /\.(pdf|xlsx|xls|csv|docx|doc)$/i;
var MAX_ATTACH_BYTES = 8 * 1024 * 1024;
var BATCH = 4;

/* ---------- one-click web app ---------- */

function doGet(e) {
  var p = (e && e.parameter) || {};
  try {
    if (p.action === 'disconnect') {
      removeTriggers();
      PropertiesService.getUserProperties().deleteProperty('connectKey');
      return page('Disconnected', 'Quote Desk will no longer read your Gmail. You can close this tab.');
    }
    var key = String(p.key || '');
    if (key.indexOf('qd_') !== 0) return page('Missing key', 'Open this page from Quote Desk > Settings > Connect Gmail.');
    PropertiesService.getUserProperties().setProperty('connectKey', key);
    installTrigger();
    var n = scanNow();
    return page('Gmail connected ✓', 'Quote Desk will check your Gmail every 5 minutes (' + n + ' email(s) sent now). You can close this tab and go back to Quote Desk.');
  } catch (err) {
    return page('Could not connect', String(err.message || err));
  }
}

function page(title, msg) {
  var html = '<!doctype html><meta name="viewport" content="width=device-width,initial-scale=1">' +
    '<div style="font-family:system-ui,sans-serif;max-width:460px;margin:15vh auto;padding:28px;border:1px solid #DDE3EC;border-radius:16px;color:#1A2433">' +
    '<div style="width:36px;height:36px;border-radius:8px;background:#0B3A6E;color:#fff;display:flex;align-items:center;justify-content:center;font-weight:700">B</div>' +
    '<h1 style="font-size:22px;margin:16px 0 8px">' + title + '</h1><p style="color:#4A5668;line-height:1.5">' + msg + '</p></div>';
  return HtmlService.createHtmlOutput(html).setTitle('Quote Desk – ' + title);
}

/* ---------- manual mode ---------- */

function setup() {
  if (CONNECT_KEY.indexOf('qd_') !== 0) throw new Error('Paste your Connect key first.');
  PropertiesService.getUserProperties().setProperty('connectKey', CONNECT_KEY);
  installTrigger();
  scanNow();
  Logger.log('Connected. Quote Desk will check your Gmail every 5 minutes.');
}

function disconnect() {
  removeTriggers();
  PropertiesService.getUserProperties().deleteProperty('connectKey');
  Logger.log('Disconnected. No more emails will be sent to Quote Desk.');
}

/* ---------- shared ---------- */

function getKey() {
  var k = PropertiesService.getUserProperties().getProperty('connectKey') || CONNECT_KEY;
  if (String(k).indexOf('qd_') !== 0) throw new Error('No connect key. Connect again from Quote Desk > Settings.');
  return k;
}

function installTrigger() {
  if (API_URL.indexOf('https://') !== 0) throw new Error('API_URL is not set in the Connector script.');
  removeTriggers();
  ScriptApp.newTrigger('scanNow').timeBased().everyMinutes(5).create();
}

function removeTriggers() {
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === 'scanNow') ScriptApp.deleteTrigger(t);
  });
}

function scanNow() {
  var key = getKey();
  var props = PropertiesService.getUserProperties();
  var seen = JSON.parse(props.getProperty('seen') || '[]');
  var seenSet = {};
  seen.forEach(function (id) { seenSet[id] = true; });
  var me = Session.getActiveUser().getEmail().toLowerCase();

  var sent = 0;

  // 1) deal scans Quote Desk asked for (new deal / "Read Gmail again")
  var tasks = [];
  try { tasks = call({ action: 'connector.tasks', connectKey: key }).tasks || []; } catch (e) { Logger.log(e.message); }
  tasks.forEach(function (t) {
    var list = [];
    GmailApp.search(t.q, 0, 25).forEach(function (thread) {
      thread.getMessages().forEach(function (m) {
        if (SKIP_SUBJECT.test(m.getSubject())) return;
        list.push(pack(m, thread, me));
      });
    });
    list = list.slice(-40);
    if (!list.length) { call({ action: 'connector.ingest', connectKey: key, dealId: t.dealId, final: true, emails: [] }); return; }
    for (var i = 0; i < list.length; i += BATCH) {
      var part = list.slice(i, i + BATCH);
      call({ action: 'connector.ingest', connectKey: key, dealId: t.dealId, final: i + BATCH >= list.length, emails: part });
      sent += part.length;
    }
  });

  // 2) new deal-like emails
  var batch = [];
  GmailApp.search(SEARCH, 0, 40).forEach(function (thread) {
    thread.getMessages().forEach(function (m) {
      var id = m.getId();
      if (seenSet[id]) return;
      seenSet[id] = true; seen.push(id);
      if (SKIP_SUBJECT.test(m.getSubject())) return;
      batch.push(pack(m, thread, me));
      if (batch.length >= BATCH) { sent += send(batch, key); batch = []; }
    });
  });
  if (batch.length) sent += send(batch, key);
  props.setProperty('seen', JSON.stringify(seen.slice(-800)));
  if (!sent) send([], key); // heartbeat so Quote Desk shows "last checked"
  Logger.log('Sent ' + sent + ' email(s) to Quote Desk.');
  return sent;
}

function pack(m, thread, me) {
  var atts = m.getAttachments({ includeInlineImages: false }).filter(function (a) {
    return ATTACH_OK.test(a.getName()) && a.getSize() <= MAX_ATTACH_BYTES;
  }).slice(0, 6).map(function (a) {
    return { name: a.getName(), mimeType: a.getContentType(), b64: Utilities.base64Encode(a.getBytes()) };
  });
  return {
    id: m.getId(), threadId: thread.getId(),
    from: m.getFrom(), to: m.getTo(), subject: m.getSubject(),
    sent: !!me && String(m.getFrom()).toLowerCase().indexOf(me) >= 0,
    date: Utilities.formatDate(m.getDate(), 'Asia/Phnom_Penh', "yyyy-MM-dd'T'HH:mm"),
    link: 'https://mail.google.com/mail/u/0/#all/' + thread.getId(),
    body: String(m.getPlainBody() || '').slice(0, 8000),
    attachments: atts
  };
}

function send(emails, key) {
  call({ action: 'connector.ingest', connectKey: key, emails: emails });
  return emails.length;
}

function call(body) {
  var res = UrlFetchApp.fetch(API_URL, {
    method: 'post', contentType: 'text/plain', muteHttpExceptions: true, followRedirects: true,
    payload: JSON.stringify(body)
  });
  var out = {};
  try { out = JSON.parse(res.getContentText()); } catch (e) { throw new Error('Quote Desk did not answer: ' + res.getResponseCode()); }
  if (!out.ok) throw new Error('Quote Desk: ' + out.error);
  return out.data || {};
}
